#include <stdio.h>

void imprimirDeUmaN (int num) {
    // 1. Imprimir os números de 1 a N.
    printf ("Digite um número: ");
    scanf ("%d", &num);
    
    for (int i = 1; i <= num; i++) {
        printf ("%d ", i);
    }
    printf ("\n");
}

void imprimirDeNaUm (int num) {
    // 2. Imprimir os números de N a 1.
    printf ("Digite um número positivo: ");
    scanf ("%d", &num);
    
    for (int i = num; i <= num; i--) {
        if (i == 0){
            break;
        }
        printf ("%d ", i);    
    }
    printf ("\n");
}
